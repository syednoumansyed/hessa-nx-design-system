A reusable component that renders form controls from a configuration array. Supports multi-column rows using the row property, helper text, and various control types.

Localized Pair Validation

Use the localizedPair property to create cross-field validation for bilingual fields. If one field has a value, the paired field becomes required.

{{ localizedPairDemoForm.value | json }}

Usage

{{ '{' }}
  type: 'input',
  formControlName: 'enDescription',
  label: 'Description (English)',
  localizedPair: 'arDescription'  // Points to the paired control
{{ '}' }},
{{ '{' }}
  type: 'input',
  formControlName: 'arDescription',
  label: 'Description (Arabic)',
  localizedPair: 'enDescription'  // Points back to complete the pair
{{ '}' }}

Note: Control names must start with en or ar for automatic error message generation.

Usage

1. Import the component 

import {{ '{' }} DsFormRendererComponent, FormControlConfig {{ '}' }} from '@shared/components/ds-form-control-generator/ds-form-renderer/ds-form-renderer.component';

2. Define your config

formConfig: FormControlConfig[] = [
  // Multi-column row (same 'row' value groups them)
  {{ '{' }} type: 'input', formControlName: 'firstName', label: 'First Name', row: 'name' {{ '}' }},
  {{ '{' }} type: 'input', formControlName: 'lastName', label: 'Last Name', row: 'name' {{ '}' }},

  // Single column (no 'row' property)
  {{ '{' }} type: 'input', formControlName: 'email', label: 'Email', helperText: 'Your email address' {{ '}' }},

  // Checkbox with icon
  {{ '{' }}
    type: 'checkbox',
    formControlName: 'privacy',
    selectValues: [{{ '{' }} value: 'privacy', displayedValue: 'Enable Privacy', icon: {{ '{' }} name: faShieldCheck, placement: 'end' {{ '}' }} {{ '}' }}]
  {{ '}' }}
];

3. Use in template

<form [formGroup]="form">
  <app-ds-form-renderer [controls]="formConfig" gap="lg" />
</form>
